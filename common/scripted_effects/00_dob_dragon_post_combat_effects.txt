# ESCOPO: chamado de on_combat_end_winner/loser
#   Root = combat_side (Nï¿½O character)
#   every_side_commander ? itera character scope de cada commander
#   is_current_dragonrider_warfare ? trigger AGOT (inclui flag agot_not_using_dragon)
#
# MORTE DE DRAGï¿½O:
#   death = { death_reason = death_battle } ? CK3 dispara on_death automaticamente
#   on_death ? AGOT dispara agot_dragon.0001 ? que notifica o rider
#   Portanto: Nï¿½O precisamos de evento prï¿½prio "dragï¿½o morreu"
#   AGOT jï¿½ faz tudo: cleanup de variï¿½veis, notificaï¿½ï¿½o, memï¿½ria de personagem
#
# PRESTï¿½GIO:
#   Dado a TODOS os riders (AI + player) que venceram com dragï¿½o
#   Fï¿½rmula: 100 base + (dragon_size * 2)
#   Exemplo: dragï¿½o size 80 ? +260 prestï¿½gio
#   Exemplo: dragï¿½o size 150 ? +400 prestï¿½gio
#
# VARIï¿½VEIS ARMAZENADAS NO RIDER (para eventos):
#   dob_dragon_size      = tamanho do dragï¿½o (armazenado ANTES da rolagem de morte)
#   dob_prestige_gained  = prestï¿½gio ganho (armazenado ANTES de aplicar)
#   dob_dragon_power     = poder de batalha calculado (para evento 003)
# ============================================================

# -- Sincroniza current_dragon a partir da relação oficial do AGOT ---------------
# Scope: rider (character)
dob_sync_current_dragon_from_relation = {
	if = {
		limit = {
			is_current_dragonrider = yes
			NOT = {
				AND = {
					exists = var:current_dragon
					var:current_dragon = { is_alive = yes has_trait = dragon }
				}
			}
		}
		random_relation = {
			type = agot_dragon
			is_alive = yes
			has_trait = dragon
			save_scope_as = dob_synced_dragon
		}
		if = {
			limit = { exists = scope:dob_synced_dragon }
			set_variable = {
				name = current_dragon
				value = scope:dob_synced_dragon
			}
		}
	}
}

# -- Efeito para o lado VENCEDOR -------------------------------------------------
# Root = combat_side (winner)
dob_dragon_winner_effect = {
	# Cobre tanto commanders quanto knights ï¿½ jogador pode estar em qualquer posiï¿½ï¿½o
	every_side_commander = {
		limit = {
			OR = {
				AND = {
					has_variable = current_dragon
					exists = var:current_dragon
					var:current_dragon = { is_alive = yes has_trait = dragon }
				}
				is_current_dragonrider = yes
			}
		}
		dob_sync_current_dragon_from_relation = yes

		# Guardar scope do dragï¿½o ANTES de qualquer rolagem de morte
		var:current_dragon = { save_scope_as = dob_event_dragon }
		save_scope_as = dob_rider

		set_variable = { name = dob_dragon_size value = dob_dragon_size_value }
		set_variable = { name = dob_prestige_gained value = dob_dragon_prestige_gain }

		# Dar prestï¿½gio a TODOS os riders que venceram (AI e player)
		add_prestige = dob_dragon_prestige_gain

		# 1% de chance de morte mesmo na vitï¿½ria
		random = {
			chance = 1
			var:current_dragon = { death = { death_reason = death_battle } }
		}

		# -- Dragï¿½o sobreviveu ----------------------------------------------
		if = {
			limit = { scope:dob_event_dragon = { is_alive = yes } }

			# Buff de experiï¿½ncia de batalha (2 anos) ï¿½ visï¿½vel na ficha do dragï¿½o

				remove_character_modifier = dob_dragon_battle_hardened

			}

			# 10% chance de ferimento mesmo na vitï¿½ria
			random = {
				chance = 10
				scope:dob_event_dragon = {
					remove_character_modifier = dob_dragon_wounded
					add_character_modifier = { modifier = dob_dragon_wounded days = 90 }
				}
				set_variable = { name = dob_dragon_was_wounded value = 1 }
			}

			if = {
				limit = { is_ai = no }
				if = {
					limit = { var:dob_dragon_size >= 150 }
					add_character_modifier = { modifier = dob_dragon_victory_large days = 30 }
				}
				else_if = {
					limit = { var:dob_dragon_size >= 80 }
					add_character_modifier = { modifier = dob_dragon_victory_medium days = 30 }
				}
				else = {
					add_character_modifier = { modifier = dob_dragon_victory_small days = 30 }
				}
				# -- Detectar: batalha ofensiva ou defensiva? -------------------------
				if = {
					limit = {
						any_character_war = {
							any_war_defender = { this = scope:dob_rider }
						}
					}
					# -- VITï¿½RIA DEFENSIVA --------------------------------------------
					trigger_event = { id = dragon_battle.009 days = 1 }
					if = {
						limit = { has_variable = dob_dragon_was_wounded }
						trigger_event = { id = dragon_battle.013 days = 2 }
						remove_variable = dob_dragon_was_wounded
					}
					# 15% chance de ganhar tï¿½tulo defensivo permanente no dragï¿½o
					# FIX: dispara evento 019 ao player quando habilidade ï¿½ adquirida
					scope:dob_event_dragon = {
						save_scope_as = dob_skillup_dragon
						random = {
							chance = 15
							modifier = {
								factor = 0
								has_character_modifier = dob_dragon_field_guardian
								has_character_modifier = dob_dragon_army_defender
								has_character_modifier = dob_dragon_shield_realm
							}
							if = {
								limit = { NOT = { has_character_modifier = dob_dragon_field_guardian } }
								add_character_modifier = dob_dragon_field_guardian
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_dragon }
									trigger_event = { id = dragon_battle.019 days = 2 }
								}
							}
							else_if = {
								limit = { NOT = { has_character_modifier = dob_dragon_army_defender } }
								add_character_modifier = dob_dragon_army_defender
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_dragon }
									trigger_event = { id = dragon_battle.019 days = 2 }
								}
							}
							else_if = {
								limit = { NOT = { has_character_modifier = dob_dragon_shield_realm } }
								add_character_modifier = dob_dragon_shield_realm
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_dragon }
									trigger_event = { id = dragon_battle.019 days = 2 }
								}
							}
						}
					}
				}
				else = {
					# -- VITï¿½RIA OFENSIVA ----------------------------------
					trigger_event = { id = dragon_battle.002 days = 1 }
					if = {
						limit = { has_variable = dob_dragon_was_wounded }
						trigger_event = { id = dragon_battle.007 days = 2 }
						remove_variable = dob_dragon_was_wounded
					}
					# Tï¿½tulos ofensivos ï¿½ rolls independentes, cada um dispara seu prï¿½prio evento
					# The Warbringer: 55% ï¿½ fï¿½cil, bï¿½nus pequeno
					scope:dob_event_dragon = {
						random = {
							chance = 55
							modifier = { factor = 0 has_character_modifier = dob_dragon_the_warbringer }
							if = {
								limit = { NOT = { has_character_modifier = dob_dragon_the_warbringer } }
								add_character_modifier = dob_dragon_the_warbringer
								save_scope_as = dob_skillup_off_dragon
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_off_dragon }

								}
							}
						}
					}
					# The Realmforged: 40% ï¿½ mï¿½dio
					scope:dob_event_dragon = {
						random = {
							chance = 40
							modifier = { factor = 0 has_character_modifier = dob_dragon_the_realmforged }
							if = {
								limit = { NOT = { has_character_modifier = dob_dragon_the_realmforged } }
								add_character_modifier = dob_dragon_the_realmforged
								save_scope_as = dob_skillup_off_dragon
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_off_dragon }
									trigger_event = { id = dragon_battle.021 days = 3 }
								}
							}
						}
					}
					# The Worldbreaker: 5% ï¿½ lendï¿½rio, difï¿½cil, bï¿½nus muito alto
					scope:dob_event_dragon = {

							chance = 5
							modifier = { factor = 0 has_character_modifier = dob_dragon_the_worldbreaker }
							if = {
								limit = { NOT = { has_character_modifier = dob_dragon_the_worldbreaker } }
								add_character_modifier = dob_dragon_the_worldbreaker
								save_scope_as = dob_skillup_off_dragon
								scope:dob_rider = {
									set_variable = { name = dob_skillup_dragon value = scope:dob_skillup_off_dragon }
									trigger_event = { id = dragon_battle.022 days = 4 }
								}
							}
						}
					}
				}
			}
		}
		# -- Dragï¿½o morreu na vitï¿½ria (1% raro) ï¿½ detectar ofensivo/defensivo ----------
		else_if = {
			limit = { is_ai = no }
			if = {
				limit = {
					any_character_war = {
						any_war_defender = { this = scope:dob_rider }
					}
				}
				trigger_event = { id = dragon_battle.011 days = 1 }
			}
			else = {
				trigger_event = { id = dragon_battle.005 days = 1 }
			}
		}
		# -- Habilidades defensivas para dragï¿½es de IA (defensores) --------------------
		# FIX: o bloco is_ai = no acima nï¿½o cobre IA. Este roda para TODOS os
		# dragï¿½es de IA que estavam defendendo ï¿½ sï¿½ modifier, sem evento.
		if = {
			limit = {
				is_ai = yes
				scope:dob_event_dragon = { is_alive = yes }
				any_character_war = {
					any_war_defender = { this = scope:dob_rider }
				}
			}
			scope:dob_event_dragon = {
				random = {
					chance = 15
					modifier = {
						factor = 0
						has_character_modifier = dob_dragon_field_guardian
						has_character_modifier = dob_dragon_army_defender
						has_character_modifier = dob_dragon_shield_realm
					}
					if = {
						limit = { NOT = { has_character_modifier = dob_dragon_field_guardian } }
						add_character_modifier = dob_dragon_field_guardian
					}
					else_if = {
						limit = { NOT = { has_character_modifier = dob_dragon_army_defender } }
						add_character_modifier = dob_dragon_army_defender
					}
					else_if = {
						limit = { NOT = { has_character_modifier = dob_dragon_shield_realm } }
						add_character_modifier = dob_dragon_shield_realm
					}
				}
			}
		}
		# -- Habilidades OFENSIVAS para dragï¿½es de IA (atacantes) ---------------------
		# Rolls independentes, sem evento (IA nï¿½o tem tela de notificaï¿½ï¿½o)
		if = {
			limit = {
				is_ai = yes
				scope:dob_event_dragon = { is_alive = yes }
				NOT = {
					any_character_war = {
						any_war_defender = { this = scope:dob_rider }
					}
				}
			}
			# Warbringer ï¿½ 55%
			scope:dob_event_dragon = {
				random = {
					chance = 55
					modifier = { factor = 0 has_character_modifier = dob_dragon_the_warbringer }
					if = {
						limit = { NOT = { has_character_modifier = dob_dragon_the_warbringer } }
						add_character_modifier = dob_dragon_the_warbringer
					}
				}
			}
			# Realmforged ï¿½ 40%
			scope:dob_event_dragon = {
				random = {
					chance = 40
					modifier = { factor = 0 has_character_modifier = dob_dragon_the_realmforged }
					if = {
						limit = { NOT = { has_character_modifier = dob_dragon_the_realmforged } }
						add_character_modifier = dob_dragon_the_realmforged
					}
				}
			}
			# Worldbreaker ï¿½ 5%
			scope:dob_event_dragon = {
				random = {
					chance = 5
					modifier = { factor = 0 has_character_modifier = dob_dragon_the_worldbreaker }
					if = {
						limit = { NOT = { has_character_modifier = dob_dragon_the_worldbreaker } }
						add_character_modifier = dob_dragon_the_worldbreaker
					}
				}
			}
		}
	}

	# -- KNIGHT vencedor com dragï¿½o ï¿½ mesmo comportamento que commander player --------
	# Cobre o caso em que o player serve como knight (nï¿½o commander) na batalha
	every_side_knight = {
		limit = {
			is_ai = no
			OR = {
				AND = {
					has_variable = current_dragon
					exists = var:current_dragon
					var:current_dragon = { is_alive = yes has_trait = dragon }
				}
				is_current_dragonrider = yes
			}
		}
		dob_sync_current_dragon_from_relation = yes
		var:current_dragon = { save_scope_as = dob_event_dragon }
		save_scope_as = dob_rider

		set_variable = { name = dob_dragon_size value = dob_dragon_size_value }
		set_variable = { name = dob_prestige_gained value = dob_dragon_prestige_gain }
		add_prestige = dob_dragon_prestige_gain

		if = {
			limit = { scope:dob_event_dragon = { is_alive = yes } }
			scope:dob_event_dragon = {
				remove_character_modifier = dob_dragon_battle_hardened
				add_character_modifier = { modifier = dob_dragon_battle_hardened days = 730 }
			}
			random = {
				chance = 10
				scope:dob_event_dragon = {
					remove_character_modifier = dob_dragon_wounded
					add_character_modifier = { modifier = dob_dragon_wounded days = 90 }
				}
				set_variable = { name = dob_dragon_was_wounded value = 1 }
			}
			# Detectar vitï¿½ria ofensiva ou defensiva para notificaï¿½ï¿½o
			if = {
				limit = {
					any_character_war = {
						any_war_defender = { this = scope:dob_rider }
					}
				}
				trigger_event = { id = dragon_battle.009 days = 1 }
				if = {
					limit = { has_variable = dob_dragon_was_wounded }
					trigger_event = { id = dragon_battle.013 days = 2 }
					remove_variable = dob_dragon_was_wounded
				}
			}
			else = {
				trigger_event = { id = dragon_battle.002 days = 1 }
				if = {
					limit = { has_variable = dob_dragon_was_wounded }
					trigger_event = { id = dragon_battle.007 days = 2 }
					remove_variable = dob_dragon_was_wounded
				}
			}
		}
		else_if = {
			limit = { is_ai = no }
			if = {
				limit = {
					any_character_war = {
						any_war_defender = { this = scope:dob_rider }
					}
				}
				trigger_event = { id = dragon_battle.011 days = 1 }
			}
			else = {
				trigger_event = { id = dragon_battle.005 days = 1 }
			}
		}
	}

	# -- Notificar player sobre dragï¿½es ALIADOS (mesma side, IA) ----------------------
	# Roda apenas se hï¿½ um player neste lado
	if = {
		limit = { any_side_commander = { is_ai = no } }
		every_side_commander = {
			limit = {
				is_ai = yes
				has_variable = current_dragon
				exists = var:current_dragon
				var:current_dragon = { is_alive = yes has_trait = dragon }
			}
			var:current_dragon = {
				save_scope_as = dob_ally_dragon
				
			}
			save_scope_as = dob_ally_rider_combat
			
			# root = combat_side (winner) ï¿½ sempre vï¿½lido em scripted_effect
			# FIX: prev era instï¿½vel quando chamado via on_action; root ï¿½ garantido
			root = {
				every_side_commander = {
					limit = { is_ai = no }
					trigger_event = { id = dragon_battle.014 days = 2 }
				}
			}
		}
	}

	# -- Notificar player sobre dragï¿½es de CAPITï¿½ES (player ausente da batalha) ------
	# Cobre vassalos/contratados do player lutando com dragï¿½o quando o player
	# nï¿½o estï¿½ pessoalmente na batalha como commander ou knight
	if = {
		limit = {
			NOT = { any_side_commander = { is_ai = no } }
			NOT = { any_side_knight    = { is_ai = no } }
			any_side_commander = {
				is_ai = yes
				has_variable = current_dragon
				exists = var:current_dragon
				var:current_dragon = { is_alive = yes has_trait = dragon }

					employer = { is_ai = no }
					top_liege = { is_ai = no }
				}
			}

		# Dragï¿½es aliados do capitï¿½o ? evento 014 ao player

			limit = {
				is_ai = yes
				has_variable = current_dragon
				exists = var:current_dragon
				var:current_dragon = { is_alive = yes has_trait = dragon }
				OR = {
					employer = { is_ai = no }
					top_liege = { is_ai = no }
				}
			}
			var:current_dragon = {
				save_scope_as = dob_ally_dragon
				
			}
			save_scope_as = dob_ally_rider_combat
			
			if = {
				limit = { employer = { is_ai = no } }
				employer = { trigger_event = { id = dragon_battle.014 days = 1 } }
			}
			else_if = {
				limit = { top_liege = { is_ai = no } }
				top_liege = { trigger_event = { id = dragon_battle.014 days = 1 } }
			}
		}
		# Dragï¿½es inimigos do capitï¿½o ? evento 015 ao player

					has_variable = current_dragon

					every_side_commander = {
						limit = {
							is_ai = yes
							OR = {
								employer = { is_ai = no }
								top_liege = { is_ai = no }
							}
						}
						if = {
							limit = { employer = { is_ai = no } }
							employer = { trigger_event = { id = dragon_battle.015 days = 3 } }
						}
						else_if = {
							limit = { top_liege = { is_ai = no } }
							top_liege = { trigger_event = { id = dragon_battle.015 days = 3 } }
						}
					}
				}
			}
		}
	}

	# -- Notificar player sobre dragï¿½es INIMIGOS (side perdedora) --------------------
	# Delay de 3 dias para garantir que loser_effect jï¿½ processou mortes/ferimentos
	if = {
		limit = { any_side_commander = { is_ai = no } }
		enemy_side = {
			every_side_commander = {
				limit = {
					has_variable = current_dragon
					exists = var:current_dragon
				var:current_dragon = { is_alive = yes has_trait = dragon }
				}
				save_scope_as = dob_enemy_rider_combat
				var:current_dragon = { save_scope_as = dob_enemy_dragon }
				# root = combat_side (winner) ï¿½ encontrar o player no lado vencedor
				root = {
					every_side_commander = {
						limit = { is_ai = no }
						trigger_event = { id = dragon_battle.015 days = 3 }
					}
				}
			}
		}
	}

	# -- Gravar participantes dragï¿½es para o Battle Report GUI ------------------------
	# Roda se o player (ou vassalo/capitï¿½o do player) estï¿½ neste lado
	if = {
		limit = {
			OR = {
				any_side_commander = { is_ai = no }
				any_side_knight    = { is_ai = no }
				any_side_commander = {
					is_ai = yes
					OR = {
						employer = { is_ai = no }
						top_liege = { is_ai = no }
					}
				}
				any_side_knight = {
					is_ai = yes
					OR = {
						employer = { is_ai = no }
						top_liege = { is_ai = no }
					}
				}
			}
		}
		dob_record_battle_dragons_effect = yes
	}
}

# -- Efeito para o lado PERDEDOR -------------------------------------------------
# Root = combat_side (loser)
dob_dragon_loser_effect = {
	every_side_commander = {
		limit = {
			OR = {
				AND = {
					has_variable = current_dragon
					exists = var:current_dragon
					var:current_dragon = { is_alive = yes has_trait = dragon }
				}
				is_current_dragonrider = yes
			}
		}
		dob_sync_current_dragon_from_relation = yes
		# Estamos agora em character scope (o rider)

		# Guardar scope do dragï¿½o ANTES da rolagem de morte
		var:current_dragon = { save_scope_as = dob_event_dragon }
		save_scope_as = dob_rider_loser

		# Armazenar ANTES da rolagem de morte
		set_variable = {
			name = dob_dragon_size
			value = dob_dragon_size_value
		}
		set_variable = {
			name = dob_dragon_power
			value = dob_dragon_battle_bonus
		}

		# 20% chance de morte na derrota
		random = {
			chance = 20
			# Penalidade de prestï¿½gio: perder um dragï¿½o ï¿½ catastrï¿½fico
			add_prestige = -300
			# Matar o dragï¿½o ? AGOT dispara on_death ? agot_dragon.0001
			# ? agot_dragon_death.0001 notifica o rider automaticamente
