# ============================================================
# DRAGON ON BATTLE - Fix MDE Broken Dragons
# Conserta dragões criados pelo More Dragon Eggs que não foram
# registrados no sistema de armazenamento do AGOT (gl_dragon_variable_storage).
#
# CAUSA RAIZ:
#   AGOT armazena dragon_size_base em uma global list (gl_dragon_variable_storage)
#   via story cycles. Dragões criados por mods que não chamam
#   agot_dragon_transfer_vars_to_story_cycle_effect ficam invisíveis para esse
#   sistema → dragon_size sempre retorna 0 → Tame/Rename desabilitados.
#
# SOLUÇÃO:
#   Detectar dragões sem entrada no storage system (var:dragon_id não encontrado
#   em nenhuma story na gl_dragon_variable_storage) e chamar a função oficial do
#   AGOT para registrá-los, estimando dragon_size_base pela idade do personagem.
#
# Scope: o DRAGÃO (has_trait = dragon)
# ============================================================

# Estima dragon_size_base baseado na idade do dragão.
# Fórmula baseada na taxa de crescimento do AGOT:
#   Age < 11: +3 size/year (young phase)
#   Age 11-15: +2 size/year (adolescent)
#   Age 15+: +~1 size/year (adult, já crescido)
# Valores aproximados, conservadores (sem traits de physique).
dob_estimate_dragon_size_from_age = {
	if = {
		limit = { age >= 80 }
		set_variable = { name = dragon_size_base value = 175 }
	}
	else_if = {
		limit = { age >= 60 }
		set_variable = { name = dragon_size_base value = 150 }
	}
	else_if = {
		limit = { age >= 45 }
		set_variable = { name = dragon_size_base value = 120 }
	}
	else_if = {
		limit = { age >= 30 }
		set_variable = { name = dragon_size_base value = 90 }
	}
	else_if = {
		limit = { age >= 20 }
		set_variable = { name = dragon_size_base value = 60 }
	}
	else_if = {
		limit = { age >= 15 }
		set_variable = { name = dragon_size_base value = 40 }
	}
	else_if = {
		limit = { age >= 10 }
		set_variable = { name = dragon_size_base value = 30 }
	}
	else_if = {
		limit = { age >= 5 }
		set_variable = { name = dragon_size_base value = 15 }
	}
	else = {
		set_variable = { name = dragon_size_base value = 5 }
	}
}

# ─────────────────────────────────────────────────────────────────────────────
# Abordagem: DELETE & RECREATE
# Para cada dragão com tamanho 0 na story:
#   1. Estima tamanho correto pela idade
#   2. Apaga a story corrompida da gl_dragon_variable_storage e a encerra
#   3. Re-registra o dragão do zero via função oficial do AGOT
#      (que lê dragon_size_base e dragon_age do personagem do dragão)
# ─────────────────────────────────────────────────────────────────────────────

# Conserta um único dragão: apaga story corrompida e recria do zero.
# Scope: o DRAGÃO (has_trait = dragon)
# REQUER: scope:dob_fix_caller já salvo (o personagem do player, para contador)
dob_fix_single_dragon_delete_recreate = {
	# Passo 1: estimar tamanho correto e setar no dragon character
	dob_estimate_dragon_size_from_age = yes

	# Garantir dragon_age no personagem (lido pelo on_setup da nova story)
	if = {
		limit = { NOT = { has_variable = dragon_age } }
		set_variable = { name = dragon_age value = { value = 0 add = age } }
	}

	# Passo 2: apagar story corrompida (se existir)
	save_temporary_scope_as = dob_current_dragon
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = { var:dragon_id ?= scope:dob_current_dragon }
		remove_list_global_variable = {
			name = gl_dragon_variable_storage
			target = this
		}
		end_story = yes
	}

	# Passo 3: registrar o dragão de forma direta, sem criar personagem temporário
	# Isso evita o problema de scope nas closures do create_character
	# O on_setup da story lê 'title:c_ruins.holder.var:transfered_dragon'
	title:c_ruins.holder = {
		set_variable = {
			name = transfered_dragon
			value = scope:dob_current_dragon
		}
	}
	# Criar a story diretamente no dragão — on_setup roda imediatamente
	create_story = {
		type = story_dragon_variable_storage
		save_scope_as = dob_new_dragon_story
	}
	# Adicionar à lista global para que o sistema AGOT encontre
	add_to_global_variable_list = {
		name = gl_dragon_variable_storage
		target = scope:dob_new_dragon_story
	}
	# Adicionar à living_dragons se ainda não estiver
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = living_dragons
					target = this
				}
			}
		}
		add_to_global_variable_list = {
			name = living_dragons
			target = this
		}
	}
	# Registrar na árvore genealógica de dragões
	agot_dragon_tree_creation_effect = { DRAGON = this }

	# Passo 4: incrementar contador no player
	scope:dob_fix_caller = {
		change_variable = { name = dob_fix_count add = 1 }
	}
}

# ─────────────────────────────────────────────────────────────────────────────
# ON LOAD: corrigir tamanho de TODOS os dragões vivos ao entrar no jogo.
# Diferente de dob_fix_all_broken_dragons (que só trata size=0 ou fora da lista),
# este effect aplica o piso mínimo de tamanho por IDADE em QUALQUER dragão cujo
# dragon_size_base esteja abaixo do esperado — independente de estar na lista.
# Também registra dragões totalmente fora da story cycle.
# Scope: nenhum (chamado de dob_on_game_start via every_living_character)
# ─────────────────────────────────────────────────────────────────────────────
dob_correct_all_dragon_sizes_on_load = {
	every_living_character = {
		limit = { has_trait = dragon }
		save_temporary_scope_as = dob_load_check_dragon

		if = {
			# Caso A: dragão totalmente fora da story cycle → registrar do zero
			limit = {
				NOT = {
					any_in_global_list = {
						variable = gl_dragon_variable_storage
						var:dragon_id ?= scope:dob_load_check_dragon
					}
				}
			}
			# Estimar size pela idade antes de registrar
			dob_estimate_dragon_size_from_age = yes
			if = {
				limit = { NOT = { has_variable = dragon_age } }
				set_variable = { name = dragon_age value = { value = 0 add = age } }
			}
			# Gerar gene vars de retrato ausentes antes de registrar na story cycle
			# (MDE não chama agot_gen_appearance_variables → portrait mostra erros no log)
			if = {
				limit = { NOT = { has_variable = gene_dragon_body_shading_template } }
				agot_gen_appearance_variables = yes
			}
			# Registrar via AGOT — cria story cycle correta com os valores acima
			agot_dragon_transfer_vars_to_story_cycle_effect = { DRAGON = this }
		}
		else = {
			# Caso B: dragão JÁ está na story cycle — aplicar piso de tamanho por idade
			# Regra: só AUMENTA, nunca diminui. Se AGOT já setou maior, mantém.

			# 1. Aplicar piso no var do personagem do dragão
			if = {
				limit = { age >= 80 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 175 } }
				set_variable = { name = dragon_size_base value = 175 }
			}
			else_if = {
				limit = { age >= 60 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 150 } }
				set_variable = { name = dragon_size_base value = 150 }
			}
			else_if = {
				limit = { age >= 45 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 120 } }
				set_variable = { name = dragon_size_base value = 120 }
			}
			else_if = {
				limit = { age >= 30 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 90 } }
				set_variable = { name = dragon_size_base value = 90 }
			}
			else_if = {
				limit = { age >= 20 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 60 } }
				set_variable = { name = dragon_size_base value = 60 }
			}
			else_if = {
				limit = { age >= 15 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 40 } }
				set_variable = { name = dragon_size_base value = 40 }
			}
			else_if = {
				limit = { age >= 10 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 30 } }
				set_variable = { name = dragon_size_base value = 30 }
			}
			else_if = {
				limit = { age >= 5 OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 15 } }
				set_variable = { name = dragon_size_base value = 15 }
			}
			else_if = {
				limit = { OR = { NOT = { has_variable = dragon_size_base } var:dragon_size_base < 5 } }
				set_variable = { name = dragon_size_base value = 5 }
			}

			# 2. Propagar o valor corrigido para a story cycle
			# AGOT: a story tem DOIS vars: dragon_size_base (armazenado) e dragon_size (exibido).
			# Precisamos atualizar AMBOS — AGOT exibe dragon_size, não dragon_size_base.
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = { var:dragon_id ?= scope:dob_load_check_dragon }
				# Atualizar dragon_size_base se abaixo do valor correto
				if = {
					limit = { var:dragon_size_base < scope:dob_load_check_dragon.var:dragon_size_base }
					set_variable = { name = dragon_size_base value = scope:dob_load_check_dragon.var:dragon_size_base }
				}
				# CRÍTICO: também atualizar dragon_size (valor exibido pelo AGOT)
				# Sem isso, a janela do dragão continua mostrando o valor antigo.
				if = {
					limit = {
						OR = {
							NOT = { has_variable = dragon_size }
							var:dragon_size < scope:dob_load_check_dragon.var:dragon_size_base
						}
					}
					set_variable = { name = dragon_size value = scope:dob_load_check_dragon.var:dragon_size_base }
				}
			}

			# 3. Atualizar modifier do rider imediatamente
			if = {
				limit = { has_variable = current_rider exists = var:current_rider }
				var:current_rider = {
					if = {
						limit = { exists = var:current_dragon }
						var:current_dragon = { dob_update_dragon_power_modifier = yes }
					}
					dob_update_rider_dragon_modifier = yes
				}
			}

			# 4. Corrigir gene vars de retrato ausentes na story cycle existente
			# MDE não chama agot_gen_appearance_variables → vars ausentes no personagem
			# → story cycle criada sem elas → portrait mostra erros no log.
			# Fix: gerar vars no personagem e propagar para a story existente.
			if = {
				limit = { NOT = { has_variable = gene_dragon_body_shading_template } }
				agot_gen_appearance_variables = yes
				every_in_global_list = {
					variable = gl_dragon_variable_storage
					limit = { var:dragon_id ?= scope:dob_load_check_dragon }
					# Cores primárias/secundárias/terciárias
					set_variable = { name = gene_dragon_primary_color_hue          value = scope:dob_load_check_dragon.var:gene_dragon_primary_color_hue }
					set_variable = { name = gene_dragon_primary_color_value        value = scope:dob_load_check_dragon.var:gene_dragon_primary_color_value }
					set_variable = { name = gene_dragon_secondary_hue              value = scope:dob_load_check_dragon.var:gene_dragon_secondary_hue }
					set_variable = { name = gene_dragon_secondary_value            value = scope:dob_load_check_dragon.var:gene_dragon_secondary_value }
					set_variable = { name = gene_dragon_tertiary_hue               value = scope:dob_load_check_dragon.var:gene_dragon_tertiary_hue }
					set_variable = { name = gene_dragon_tertiary_value             value = scope:dob_load_check_dragon.var:gene_dragon_tertiary_value }
					# Olhos e chifres (cor)
					set_variable = { name = gene_dragon_eye_color_hue              value = scope:dob_load_check_dragon.var:gene_dragon_eye_color_hue }
					set_variable = { name = gene_dragon_eye_color_value            value = scope:dob_load_check_dragon.var:gene_dragon_eye_color_value }
					set_variable = { name = gene_dragon_horn_color_hue             value = scope:dob_load_check_dragon.var:gene_dragon_horn_color_hue }
					set_variable = { name = gene_dragon_horn_color_value           value = scope:dob_load_check_dragon.var:gene_dragon_horn_color_value }
					# Forma do corpo / cabeça
					set_variable = { name = gene_dragon_brow_width                 value = scope:dob_load_check_dragon.var:gene_dragon_brow_width }
					set_variable = { name = gene_dragon_cheek_width                value = scope:dob_load_check_dragon.var:gene_dragon_cheek_width }
					set_variable = { name = gene_dragon_chin_profile               value = scope:dob_load_check_dragon.var:gene_dragon_chin_profile }
					set_variable = { name = gene_dragon_crest_depth                value = scope:dob_load_check_dragon.var:gene_dragon_crest_depth }
					set_variable = { name = gene_dragon_head_roundness             value = scope:dob_load_check_dragon.var:gene_dragon_head_roundness }
					set_variable = { name = gene_dragon_jaw_width                  value = scope:dob_load_check_dragon.var:gene_dragon_jaw_width }
					set_variable = { name = gene_dragon_lower_jaw_height           value = scope:dob_load_check_dragon.var:gene_dragon_lower_jaw_height }
					set_variable = { name = gene_dragon_lower_jaw_width            value = scope:dob_load_check_dragon.var:gene_dragon_lower_jaw_width }
					set_variable = { name = gene_dragon_upper_jaw_width            value = scope:dob_load_check_dragon.var:gene_dragon_upper_jaw_width }
					set_variable = { name = gene_dragon_outer_brow_height          value = scope:dob_load_check_dragon.var:gene_dragon_outer_brow_height }
					# Focinho / snout
					set_variable = { name = gene_dragon_snout_height               value = scope:dob_load_check_dragon.var:gene_dragon_snout_height }
					set_variable = { name = gene_dragon_snout_length               value = scope:dob_load_check_dragon.var:gene_dragon_snout_length }
					set_variable = { name = gene_dragon_snout_profile              value = scope:dob_load_check_dragon.var:gene_dragon_snout_profile }
					set_variable = { name = gene_dragon_snout_width                value = scope:dob_load_check_dragon.var:gene_dragon_snout_width }
					set_variable = { name = gene_dragon_snout_end_width            value = scope:dob_load_check_dragon.var:gene_dragon_snout_end_width }
					# Chifres e cristas
					set_variable = { name = gene_dragon_horns_eyebrow_length       value = scope:dob_load_check_dragon.var:gene_dragon_horns_eyebrow_length }
					set_variable = { name = gene_dragon_horns_eyebrow              value = scope:dob_load_check_dragon.var:gene_dragon_horns_eyebrow }
					set_variable = { name = gene_dragon_main_horn_shape_template   value = scope:dob_load_check_dragon.var:gene_dragon_main_horn_shape_template }
					set_variable = { name = gene_dragon_main_horn_shape_value      value = scope:dob_load_check_dragon.var:gene_dragon_main_horn_shape_value }
					# Sombreamento corpo e asas
					set_variable = { name = gene_dragon_body_shading_template      value = scope:dob_load_check_dragon.var:gene_dragon_body_shading_template }
					set_variable = { name = gene_dragon_body_shading_value         value = scope:dob_load_check_dragon.var:gene_dragon_body_shading_value }
					set_variable = { name = gene_dragon_wings_shading_template     value = scope:dob_load_check_dragon.var:gene_dragon_wings_shading_template }
					set_variable = { name = gene_dragon_wings_shading_value        value = scope:dob_load_check_dragon.var:gene_dragon_wings_shading_value }
					# Espinhos, barbatanas, cauda, pescoço
					set_variable = { name = gene_dragon_center_fin_size            value = scope:dob_load_check_dragon.var:gene_dragon_center_fin_size }
					set_variable = { name = gene_dragon_back_spike_size            value = scope:dob_load_check_dragon.var:gene_dragon_back_spike_size }
					set_variable = { name = gene_dragon_neck_spike_size            value = scope:dob_load_check_dragon.var:gene_dragon_neck_spike_size }
					set_variable = { name = gene_dragon_side_fin_size              value = scope:dob_load_check_dragon.var:gene_dragon_side_fin_size }
					set_variable = { name = gene_dragon_neck_length                value = scope:dob_load_check_dragon.var:gene_dragon_neck_length }
					set_variable = { name = gene_dragon_old_neck                   value = scope:dob_load_check_dragon.var:gene_dragon_old_neck }
					set_variable = { name = gene_dragon_tail_length                value = scope:dob_load_check_dragon.var:gene_dragon_tail_length }
					# Escamas metálicas e ferimento
					set_variable = { name = gene_dragon_metallic_scales_strength   value = scope:dob_load_check_dragon.var:gene_dragon_metallic_scales_strength }
					set_variable = { name = gene_dragon_wounded                    value = scope:dob_load_check_dragon.var:gene_dragon_wounded }
				}
			}
		}
	}
}

# Conserta TODOS os dragões vivos com dragon_size 0 na story.
# Scope: o player (chamado da decisão)
dob_fix_all_broken_dragons = {
	# Salva o player como scope nomeado para o contador
	save_scope_as = dob_fix_caller
	set_variable = { name = dob_fix_count value = 0 }

	every_living_character = {
		limit = { has_trait = dragon }
		save_temporary_scope_as = current_dob_dragon_check
		if = {
			limit = {
				OR = {
					# Dragão está na lista com tamanho 0 na story
					any_in_global_list = {
						variable = gl_dragon_variable_storage
						var:dragon_id ?= scope:current_dob_dragon_check
						var:dragon_size_base <= 0
					}
					# Dragão não está na lista (mod sem registro)
					NOT = {
						any_in_global_list = {
							variable = gl_dragon_variable_storage
							var:dragon_id ?= scope:current_dob_dragon_check
						}
					}
				}
			}
			dob_fix_single_dragon_delete_recreate = yes
		}
		else = {
		}
	}
}
