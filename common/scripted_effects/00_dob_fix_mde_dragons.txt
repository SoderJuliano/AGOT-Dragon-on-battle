# ============================================================
# DRAGON ON BATTLE - Fix MDE Broken Dragons
# Conserta dragões criados pelo More Dragon Eggs que não foram
# registrados no sistema de armazenamento do AGOT (gl_dragon_variable_storage).
#
# CAUSA RAIZ:
#   AGOT armazena dragon_size_base em uma global list (gl_dragon_variable_storage)
#   via story cycles. Dragões criados por mods que não chamam
#   agot_dragon_transfer_vars_to_story_cycle_effect ficam invisíveis para esse
#   sistema → dragon_size sempre retorna 0 → Tame/Rename desabilitados.
#
# SOLUÇÃO:
#   Detectar dragões sem entrada no storage system (var:dragon_id não encontrado
#   em nenhuma story na gl_dragon_variable_storage) e chamar a função oficial do
#   AGOT para registrá-los, estimando dragon_size_base pela idade do personagem.
#
# Scope: o DRAGÃO (has_trait = dragon)
# ============================================================

# Estima dragon_size_base baseado na idade do dragão.
# Fórmula baseada na taxa de crescimento do AGOT:
#   Age < 11: +3 size/year (young phase)
#   Age 11-15: +2 size/year (adolescent)
#   Age 15+: +~1 size/year (adult, já crescido)
# Valores aproximados, conservadores (sem traits de physique).
# Tiers reais AGOT: 0=<30 | 1=30-60 | 2=60-90 | 3=90-120 | 4=120-150 | 5=150-180
#               6=180-210 | 7=210-240 | 8=240-270 | 9=270-300 | 10=300+
# Floors de EMERGÊNCIA: apenas para dragões quebrados (size=0 ou MDE não-registrado).
# O AGOT cresce dragões naturalmente via birthday events. Esses pisos SÓ devem
# interferir se o size_base for impossívelmente baixo para a idade.
# Com physique_good_3 (x1.5): age80→90x1.5=135 (Tier4); age55→45x1.5=67 (Tier2).
dob_estimate_dragon_size_from_age = {
	if = {
		limit = { age >= 150 }
		set_variable = { name = dragon_size_base value = 200 }
	}
	else_if = {
		limit = { age >= 120 }
		set_variable = { name = dragon_size_base value = 150 }
	}
	else_if = {
		limit = { age >= 100 }
		set_variable = { name = dragon_size_base value = 120 }
	}
	else_if = {
		limit = { age >= 80 }
		set_variable = { name = dragon_size_base value = 90 }
	}
	else_if = {
		limit = { age >= 65 }
		set_variable = { name = dragon_size_base value = 60 }
	}
	else_if = {
		limit = { age >= 55 }
		set_variable = { name = dragon_size_base value = 45 }
	}
	else_if = {
		limit = { age >= 45 }
		set_variable = { name = dragon_size_base value = 30 }
	}
	else_if = {
		limit = { age >= 35 }
		set_variable = { name = dragon_size_base value = 20 }
	}
	else_if = {
		limit = { age >= 25 }
		set_variable = { name = dragon_size_base value = 15 }
	}
	else_if = {
		limit = { age >= 20 }
		set_variable = { name = dragon_size_base value = 10 }
	}
	else_if = {
		limit = { age >= 15 }
		set_variable = { name = dragon_size_base value = 6 }
	}
	else_if = {
		limit = { age >= 10 }
		set_variable = { name = dragon_size_base value = 3 }
	}
	else_if = {
		limit = { age >= 5 }
		set_variable = { name = dragon_size_base value = 2 }
	}
	else = {
		set_variable = { name = dragon_size_base value = 1 }
	}
}

# ─────────────────────────────────────────────────────────────────────────────
# Abordagem: DELETE & RECREATE
# Para cada dragão com tamanho 0 na story:
#   1. Estima tamanho correto pela idade
#   2. Apaga a story corrompida da gl_dragon_variable_storage e a encerra
#   3. Re-registra o dragão do zero via função oficial do AGOT
#      (que lê dragon_size_base e dragon_age do personagem do dragão)
# ─────────────────────────────────────────────────────────────────────────────

# Conserta um único dragão: apaga story corrompida e recria do zero.
# Scope: o DRAGÃO (has_trait = dragon)
# REQUER: scope:dob_fix_caller já salvo (o personagem do player, para contador)
dob_fix_single_dragon_delete_recreate = {
	# Passo 1: estimar tamanho correto e setar no dragon character
	dob_estimate_dragon_size_from_age = yes

	# Garantir dragon_age no personagem (lido pelo on_setup da nova story)
	if = {
		limit = { NOT = { has_variable = dragon_age } }
		set_variable = { name = dragon_age value = { value = 0 add = age } }
	}

	# Passo 2: apagar story corrompida (se existir)
	save_temporary_scope_as = dob_current_dragon
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = { var:dragon_id ?= scope:dob_current_dragon }
		remove_list_global_variable = {
			name = gl_dragon_variable_storage
			target = this
		}
		# NÃO chamar end_story aqui — AGOT pode já ter encerrado esta story
		# (dragon morreu → AGOT faz end_story → nós fazemos end_story → crash double-free)
	}

	# Passo 3: registrar o dragão de forma direta, sem criar personagem temporário
	# Isso evita o problema de scope nas closures do create_character
	# O on_setup da story lê 'title:c_ruins.holder.var:transfered_dragon'
	title:c_ruins.holder = {
		set_variable = {
			name = transfered_dragon
			value = scope:dob_current_dragon
		}
	}
	# Criar a story diretamente no dragão — on_setup roda imediatamente
	create_story = {
		type = story_dragon_variable_storage
		save_scope_as = dob_new_dragon_story
	}
	# Adicionar à lista global para que o sistema AGOT encontre
	add_to_global_variable_list = {
		name = gl_dragon_variable_storage
		target = scope:dob_new_dragon_story
	}
	# Adicionar à living_dragons se ainda não estiver
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = living_dragons
					target = this
				}
			}
		}
		add_to_global_variable_list = {
			name = living_dragons
			target = this
		}
	}
	# Registrar na árvore genealógica de dragões
	agot_dragon_tree_creation_effect = { DRAGON = this }

	# Passo 4: incrementar contador no player
	scope:dob_fix_caller = {
		change_variable = { name = dob_fix_count add = 1 }
	}
}

# ─────────────────────────────────────────────────────────────────────────────
# ON LOAD: corrigir tamanho de TODOS os dragões vivos ao entrar no jogo.
# Diferente de dob_fix_all_broken_dragons (que só trata size=0 ou fora da lista),
# este effect aplica o piso mínimo de tamanho por IDADE em QUALQUER dragão cujo
# dragon_size_base esteja abaixo do esperado — independente de estar na lista.
# Também registra dragões totalmente fora da story cycle.
# Scope: nenhum (chamado de dob_on_game_start via every_living_character)
# ─────────────────────────────────────────────────────────────────────────────
dob_correct_all_dragon_sizes_on_load = {
	# Passo 0: remover story nodes de dragões mortos/inválidos.
	# Necessário porque AGOT itera gl_dragon_variable_storage para eventos de manutenção
	# (bond schemes etc.) e crasheia quando tenta abrir o GUI para personagens mortos.
	# NÃO chamar end_story aqui — AGOT pode já ter encerrado a story do dragão morto.
	# Chamar end_story em story já encerrada = acesso a ponteiro inválido = crash.
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = {
			NOT = { var:dragon_id ?= { is_alive = yes } }
		}
		remove_list_global_variable = {
			name = gl_dragon_variable_storage
			target = this
		}
	}

	# Abordagem simplificada: iterar story nodes diretamente.
	# "this" já é o story node → var:dragon_id = dragão, var:dragon_size_base = base size.
	# Sem complexidade de save_scope_as aninhado. Só AUMENTA, nunca diminui.
	# Tiers reais: 0=<30 | 1=30 | 2=60 | 3=90 | 4=120 | 5=150 | 6=180 | 7=210 | 8=240 | 9=270 | 10=300
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = {
			var:dragon_id ?= { is_alive = yes }	# ?= seguro: ghost/morto/inexistente → false
		}
		# this = story node; var:dragon_id = character do dragão; var:dragon_size_base = tamanho base
		# Floors de EMERGÊNCIA — só interferem se size_base for impossívelmente baixo.
		# Com physique_good_3 (x1.5): age150→200x1.5=300 (T10); age80→90x1.5=135 (T4); age55→45x1.5=67 (T2)
		if = {
			limit = { var:dragon_id = { age >= 150 } var:dragon_size_base < 200 }
			set_variable = { name = dragon_size_base value = 200 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 120 } var:dragon_size_base < 150 }
			set_variable = { name = dragon_size_base value = 150 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 100 } var:dragon_size_base < 120 }
			set_variable = { name = dragon_size_base value = 120 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 80 } var:dragon_size_base < 90 }
			set_variable = { name = dragon_size_base value = 90 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 65 } var:dragon_size_base < 60 }
			set_variable = { name = dragon_size_base value = 60 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 55 } var:dragon_size_base < 45 }
			set_variable = { name = dragon_size_base value = 45 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 45 } var:dragon_size_base < 30 }
			set_variable = { name = dragon_size_base value = 30 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 35 } var:dragon_size_base < 20 }
			set_variable = { name = dragon_size_base value = 20 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 25 } var:dragon_size_base < 15 }
			set_variable = { name = dragon_size_base value = 15 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 20 } var:dragon_size_base < 10 }
			set_variable = { name = dragon_size_base value = 10 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 15 } var:dragon_size_base < 6 }
			set_variable = { name = dragon_size_base value = 6 }
		}
		else_if = {
			limit = { var:dragon_id = { age >= 5 } var:dragon_size_base < 3 }
			set_variable = { name = dragon_size_base value = 3 }
		}
	}
	# Caso A: dragões fora da story cycle (MDE sem registro) — usar every_living_dragon como fallback
	every_living_dragon = {
		limit = {
			NOT = {
				any_in_global_list = {
					variable = gl_dragon_variable_storage
					var:dragon_id ?= root
				}
			}
		}
		dob_estimate_dragon_size_from_age = yes
		if = {
			limit = { NOT = { has_variable = dragon_age } }
			set_variable = { name = dragon_age value = { value = 0 add = age } }
		}
		if = {
			limit = { NOT = { has_variable = gene_dragon_body_shading_template } }
			agot_gen_appearance_variables = yes
		}
		agot_dragon_transfer_vars_to_story_cycle_effect = { DRAGON = this }
	}
}

# Conserta TODOS os dragões vivos com dragon_size 0 na story.
# Scope: o player (chamado da decisão)
dob_fix_all_broken_dragons = {
	# Salva o player como scope nomeado para o contador
	save_scope_as = dob_fix_caller
	set_variable = { name = dob_fix_count value = 0 }

	# Caso 0: LIMPAR story nodes de dragões mortos/inválidos da gl_dragon_variable_storage.
	# Dragões mortos na lista fazem o AGOT tentar processar eventos para personagens inválidos
	# → crash no GUI. Apenas remover da lista — NÃO chamar end_story.
	# end_story em story já encerrada pelo AGOT = double-free = acesso a ponteiro inválido = crash.
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = {
			NOT = { var:dragon_id ?= { is_alive = yes } }
		}
		remove_list_global_variable = {
			name = gl_dragon_variable_storage
			target = this
		}
	}

	# FIX: iterate story nodes directly — funciona de qualquer scope
	# Caso 1: dragões na story com size_base = 0 → recriar
	every_in_global_list = {
		variable = gl_dragon_variable_storage
		limit = {
			var:dragon_id ?= { is_alive = yes }	# ?= seguro: ghost/morto/inexistente → false
			var:dragon_size_base <= 0
		}
		var:dragon_id = { dob_fix_single_dragon_delete_recreate = yes }
	}
	# Caso 2: dragões fora da story (MDE sem registro) → registrar
	every_living_dragon = {
		limit = {
			NOT = {
				any_in_global_list = {
					variable = gl_dragon_variable_storage
					var:dragon_id ?= root
				}
			}
		}
		dob_fix_single_dragon_delete_recreate = yes
	}
}
