# ── Hook 1: Guerra declarada ─────────────────────────────────
# Scope: root = character que declarou a guerra (CORRETO, é character)
on_war_started = {
	on_actions = { dob_on_war_started }
}

dob_on_war_started = {
	effect = {
		# FIX: on_war_started ROOT = casus_belli em CK3+AGOT, não character!
		# Para acessar o atacante/defensor, usar scope:attacker e scope:defender.

		# Caso 1: O PLAYER (atacante) declarou guerra e tem um dragão vivo
		if = {
			limit = {
				scope:attacker = {
					is_ai = no
					OR = {
						is_current_dragonrider = yes
						AND = {
							has_variable = current_dragon
							exists = var:current_dragon
							var:current_dragon = { is_alive = yes has_trait = dragon }
						}
					}
				}
			}
			scope:attacker = {
				set_variable = {
					name = dob_dragon_power
					value = dob_dragon_battle_bonus
				}
				set_variable = {
					name = dob_dragon_size
					value = dob_dragon_size_value
				}
				trigger_event = {
					id = dragon_battle.001
					days = 1
				}
			}
		}
		# Caso 2: IA (atacante) declarou guerra com dragão → notifica player familiar/liege
		else_if = {
			limit = {
				scope:attacker = {
					is_ai = yes
					OR = {
						is_current_dragonrider = yes
						AND = {
							has_variable = current_dragon
							exists = var:current_dragon
							var:current_dragon = { is_alive = yes has_trait = dragon }
						}
					}
				}
			}
			scope:attacker = {
				set_variable = {
					name = dob_dragon_size
					value = dob_dragon_size_value
				}
				set_variable = {
					name = dob_dragon_power
					value = dob_dragon_battle_bonus
				}
				save_scope_as = dob_ally_rider
				every_close_family_member = {
					limit = { is_ai = no }
					trigger_event = { id = dragon_battle.004 }
				}
				if = {
					limit = {
						exists = liege
						liege = { is_ai = no }
					}
					liege = {
						trigger_event = { id = dragon_battle.004 }
					}
				}
			}
		}

		# Caso 3: Player está sendo ATACADO (é o DEFENSOR) e tem dragão
		if = {
			limit = {
				scope:defender = {
					is_ai = no
					is_current_dragonrider = yes
				}
			}
			scope:defender = {
				set_variable = {
					name = dob_dragon_power
					value = dob_dragon_battle_bonus
				}
				set_variable = {
					name = dob_dragon_size
					value = dob_dragon_size_value
				}
				trigger_event = {
					id = dragon_battle.008
					days = 1
				}
			}
		}
	}
}

# scope:war = a guerra que entrou

	on_actions = { dob_on_join_war_as_secondary }
}

dob_on_join_war_as_secondary = {
	effect = {
		# Caso 1: Player entrou no lado atacante com dragão
		# FIX: OR com var:current_dragon como fallback para dragões do MDE
		if = {
			limit = {
				is_ai = no
				OR = {
					is_current_dragonrider = yes
					AND = {
						has_variable = current_dragon
						exists = var:current_dragon
						var:current_dragon = { is_alive = yes has_trait = dragon }
					}
				}
				scope:war = { is_attacker = root }
			}
			set_variable = { name = dob_dragon_power value = dob_dragon_battle_bonus }
			set_variable = { name = dob_dragon_size  value = dob_dragon_size_value }
			trigger_event = { id = dragon_battle.001 days = 1 }
		}
		# Caso 2: Player entrou no lado defensor com dragão
		# FIX: OR com var:current_dragon como fallback para dragões do MDE
		else_if = {
			limit = {
				is_ai = no
				OR = {
					is_current_dragonrider = yes
					AND = {
						has_variable = current_dragon
						exists = var:current_dragon
						var:current_dragon = { is_alive = yes has_trait = dragon }
					}
				}
				NOT = { scope:war = { is_attacker = root } }
			}
			set_variable = { name = dob_dragon_power value = dob_dragon_battle_bonus }
			set_variable = { name = dob_dragon_size  value = dob_dragon_size_value }
			trigger_event = { id = dragon_battle.008 days = 1 }
		}
	}
}

# ── Hook 4: Game start — aplica modifiers em TODOS os dragonriders ao carregar ───────

on_game_start_after_lobby = {
	on_actions = { dob_on_game_start }
}

dob_on_game_start = {
	effect = {
		# NADA aqui intencionalmente.
		# on_game_start_after_lobby dispara DURANTE a inicialização do mundo,
		# antes de gl_dragon_variable_storage e story cycles estarem prontos.
		# Qualquer acesso a dragon_size, agot_dragon_transfer_vars ou global lists
		# nesse contexto causa crash imediato.

	}

# ── Hook 5b: Coroação / Sucessão — limpa story nodes ANTES do AGOT processar bond schemes ──
# Root = personagem que ganhou o título.
# AGOT dispara agot_dragon_maintenance.0101 (Bond Scheme) quando um novo rei é coroado.
# Este hook tem PRIORIDADE sobre os hooks do AGOT porque nosso arquivo "00_dob_"
# ordena alfabeticamente ANTES de "agot_" — logo o cleanup roda ANTES do bond scheme.
# Isso resolve o crash quando um dragão morto (ex: char 287337) ainda está na lista.
on_title_gain = {
	on_actions = { dob_on_title_gain_cleanup }
}

on_new_holder = {
	on_actions = { dob_on_title_gain_cleanup }
}

dob_on_title_gain_cleanup = {
	effect = {
		# Filtra apenas para reinos e impérios (onde bond schemes são ativados na coroação)
		# Evita performance hit em cada condado/baronia que muda de dono
		if = {
			limit = {
				OR = {
					scope:title = { tier >= tier_kingdom }
					# Fallback: sem scope:title disponível (on_new_holder pode não ter)
					NOT = { exists = scope:title }
				}
			}
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = {
					OR = {
						NOT = { exists = var:dragon_id }
						var:dragon_id = { is_alive = no }
					}
				}
				# NÃO chamar end_story — AGOT pode já ter encerrado esta story (dragon morte)
				# end_story em story já encerrada = ponteiro inválido = EXCEPTION_ACCESS_VIOLATION
				remove_list_global_variable = { name = gl_dragon_variable_storage target = this }
			}
		}
	}
}

# ── Hook 5c: Morte de personagem — limpa story node do dragão morto IMEDIATAMENTE ──
# Root = personagem morto. Sem esse hook, o story node do dragão morto permanece em
# gl_dragon_variable_storage e o AGOT's agot_dragon_maintenance.0101 continua a tentar
# processar o dragão inválido, causando erros em cascata e crash
# ("Could not find widget 'dragon_character_event'").
on_character_death = {
	on_actions = { dob_on_character_death }
}

dob_on_character_death = {
	effect = {
		# Só processa dragões — personagens normais não têm story node em gl_dragon_variable_storage
		if = {
			limit = { has_trait = dragon }
			# Remove TODOS os story nodes de dragões mortos da lista global
			# (cobre também outros dragões mortos que possam ter ficado por lá)
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = {
					OR = {
						NOT = { exists = var:dragon_id }
						var:dragon_id = { is_alive = no }
					}
				}
				# NÃO chamar end_story — AGOT pode já ter encerrado esta story (sua morte)
				# end_story em story já encerrada = double-free = EXCEPTION_ACCESS_VIOLATION
				remove_list_global_variable = { name = gl_dragon_variable_storage target = this }
			}
		}
	}
}

# ── Hook 5d: Fim de guerra — limpa lista ANTES do AGOT iterar gl_dragon_variable_storage ──
# AGOT dispara eventos de manutenção pós-guerra que ITERAM gl_dragon_variable_storage.
# Se um dragão morreu em batalha E a guerra acabou, há uma corrida de eventos:
# - AGOT enqueia evento pós-guerra que itera a lista
# - nosso on_character_death pode já ter removido o dragão, MAS a ordem não é garantida
# Solução: hook on_war_ended roda ANTES (arquivo 00_dob_ < agot_) e limpa a lista.
# Root = personagem vencedor/perdedor. Scope = qualquer personagem ativo.
on_war_ended_winner = {
	on_actions = { dob_on_war_ended_cleanup }
}

on_war_ended_loser = {
	on_actions = { dob_on_war_ended_cleanup }
}

dob_on_war_ended_cleanup = {
	effect = {
		every_in_global_list = {
			variable = gl_dragon_variable_storage
			limit = {
				OR = {
					NOT = { exists = var:dragon_id }
					var:dragon_id = { is_alive = no }
				}
			}
			# Apenas remove da lista — NÃO end_story (AGOT pode já ter encerrado)
			remove_list_global_variable = { name = gl_dragon_variable_storage target = this }
		}
	}
}

# ── Hook 5: Pulso mensal — atualiza modifier de poder na ficha do rider e do dragão ──
# Root = character. Mensal para aparecer logo após instalar o mod em save existente.
# - Dragonrider com dragão vivo: aplica/atualiza modifiers no rider E no dragão
# - Perdeu o dragão: remove os modifiers do rider automaticamente
on_monthly_pulse = {
	on_actions = { dob_on_monthly_pulse }
}

# Backup anual — garante remoção mesmo se monthly_pulse falhar
on_yearly_pulse = {
	on_actions = { dob_on_yearly_pulse }
}

dob_on_yearly_pulse = {
	effect = {
		# ── Dragonwidowed: backup anual para qualquer caso restante ──────────────────
		# Apenas remove o trait — NÃO toca dragões de AI para evitar crash no
		# More Dragon Eggs (current_rider access violation via every_living_character).
		every_living_character = {
			limit = { has_trait = dragonwidowed }
			remove_trait = dragonwidowed
		}

		# ── FIX: More Dragon Eggs crash (var:current_rider not set) ──────────────────
		# Previne crash ao setar current_rider em dragões que não têm essa variável.
		# CRÍTICO: usa has_variable (não exists) para não sobrescrever referências a riders
		# ghost/inválidos — exists falha para objetos ghost, has_variable checa só a chave.
		# Roda APENAS em dragões do player ou de seu realm para evitar problemas.
		every_living_dragon = {
			limit = {
				OR = {
					liege = root	# Dragão do player ou de seus vassals
					host = root		# Dragão em dragonpit do player
				}
				NOT = { has_variable = current_rider }	# has_variable ≠ exists — não quebra riders ghost
			}
			# Setar current_rider como 0 (null) para prevenir crashes
			set_variable = { name = current_rider value = 0 }
		}
	}
}

dob_on_monthly_pulse = {
	effect = {
		# ── PURGE dead dragons from gl_dragon_variable_storage ─────────────────────────
		# CRÍTICO: roda para TODOS os personagens (não só player) para garantir que a
		# lista está limpa ANTES de qualquer evento de manutenção do AGOT neste mesmo tick.
		# O arquivo 00_dob_on_actions.txt carrega antes de agot_* (ordem alfabética),
		# portanto este hook roda ANTES dos story events do AGOT no mesmo on_monthly_pulse.
		# Barato: só itera gl_dragon_variable_storage que normalmente está limpa.
		# NÃO chama end_story (AGOT faz isso no seu próprio on_character_death handler).
		every_in_global_list = {
			variable = gl_dragon_variable_storage
			limit = {
				OR = {
					NOT = { exists = var:dragon_id }
					var:dragon_id = { is_alive = no }
				}
			}
			remove_list_global_variable = { name = gl_dragon_variable_storage target = this }
		}

		# ── FIX ON LOAD (todo mês, só para o player) ───────────────────────────────────
		# on_monthly_pulse dispara para CADA personagem vivo como root. Sem is_ai = no,
		# dob_correct_all_dragon_sizes_on_load rodaria milhares de vezes no mesmo mês.
		# is_ai = no garante que o gatilho seja puxado UMA vez (pelo player).
		#
		# FIX COMPLETO (a cada 60 dias): dob_fix_all_broken_dragons faz delete+recreate
		# da story cycle — necessário porque dob_correct_all_dragon_sizes_on_load sozinho
		# não resolve dragões cuja story cycle foi recriada pelo AGOT com valores errados.
		# Na prática, roda no primeiro mês após qualquer carregamento (flag inexistente)
		# e depois a cada ~60 dias como garantia.
		if = {
			limit = {
				is_ai = no
				NOT = { has_character_flag = dob_fix_all_ran_recently }
			}
			add_character_flag = { flag = dob_fix_all_ran_recently days = 60 }
			dob_fix_all_broken_dragons = yes
		}

		if = {
			limit = { is_ai = no }
			dob_correct_all_dragon_sizes_on_load = yes
		}

		# ── Skybreaker: aplicar habilidade em dragões Tier 10 (dragon_size >= 300) ────
		# Usa every_living_dragon para pegar TODOS os dragões vivos, incluindo MDE
		# que podem não estar em gl_dragon_variable_storage.
		# Tier 10 = dragon_size >= 300 (level_ten_dragon_size_camera = 300 no AGOT).
		# permanent = yes: sem expiração, NOT check garante que não duplica.
		if = {
			limit = { is_ai = no }
			every_living_dragon = {
				limit = {
					dragon_size >= 300
					NOT = { has_character_modifier = dob_dragon_skybreaker }
				}
				add_character_modifier = dob_dragon_skybreaker
			}
		}

		# ── Player: atualiza modifier do dragão na ficha do rider ──────────────────────
		# Root = player character. Só toca o próprio player para evitar acessar
		# current_rider de outros dragões via hooks de outros mods (crash).
		if = {
			# Usa is_current_dragonrider (relação AGOT) em vez de exists = var:current_dragon
			# is_current_dragonrider verifica: vivo, não incapaz, dragonrider trait, relação agot_dragon ativa
			# Funciona mesmo quando var:current_dragon aponta para personagem ghost/inválido
			limit = { is_current_dragonrider = yes }
			if = {
				# FIX: is_alive = yes guards against ghost/weak references where exists = yes but char is dead
				limit = { exists = var:current_dragon  var:current_dragon = { is_alive = yes } }
				var:current_dragon = { dob_update_dragon_power_modifier = yes }
			}
			dob_update_rider_dragon_modifier = yes
		}
		else_if = {
			limit = {
				OR = {
					has_character_modifier = dob_rider_dragon_young
					has_character_modifier = dob_rider_dragon_mature
					has_character_modifier = dob_rider_dragon_great
				}
			}
			remove_character_modifier = dob_rider_dragon_young
			remove_character_modifier = dob_rider_dragon_mature
			remove_character_modifier = dob_rider_dragon_great
		}

		# ── Dragonwidowed: remover do player ─────────────────────────────────────────
		if = {
			limit = { has_trait = dragonwidowed }
			remove_trait = dragonwidowed
		}
	}
}

# ── Hook: Aniversário do dragão ──────────────────────────────
# CRÍTICO: NÃO definir effect = {} aqui! Isso substituiria o effect do AGOT
# (agot_apply_dragon_aging_effect) e impediria o crescimento do dragão.
# Em CK3, múltiplos arquivos podem definir o mesmo on_action, mas apenas
# UM effect = {} roda (o último mod carregado vence). Usar on_actions = {} 
# que é ACUMULATIVO — todos os mods que adicionam on_actions rodam.
on_dragon_birthday = {
	on_actions = { dob_on_dragon_birthday }
}

# Sub-ação do DOB para o aniversário — roda JUNTO com o agot_apply_dragon_aging_effect
# Scope: root = o DRAGÃO
# Padrão AGOT (idêntico ao change_dragon_size):
#   1. save_scope_as no dragão
#   2. every_in_global_list: save_temporary_scope_as no story node
#   3. FORA do loop: usa scope:dob_sn para ler/escrever dragon_size_base
# Não faz delete+recreate — evita erro de gene vars não encontradas.
dob_on_dragon_birthday = {
	effect = {
		if = {
			limit = { has_trait = dragon }
			save_scope_as = dob_bday_d						# root = dragão
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = { var:dragon_id ?= scope:dob_bday_d }
				save_temporary_scope_as = dob_sn			# dob_sn = story node
			}
			# Aplica piso FORA do loop (padrão AGOT). Só existe se dragão está na lista.
			if = {
				limit = { exists = scope:dob_sn }
				scope:dob_sn = {
					# this = story node, scope:dob_bday_d = dragão. Só AUMENTA nunca diminui.
					# Floors de EMERGÊNCIA — só interferem se size_base for impossívelmente baixo.
					# O AGOT cresce dragões via birthday events; nosso floor só corrige dragões quebrados.
					# Com physique_good_3 (x1.5): age150→300(T10); age80→135(T4); age55→67(T2); age45→45(T1)
					if = {
						limit = { scope:dob_bday_d = { age >= 150 } var:dragon_size_base < 200 }
						set_variable = { name = dragon_size_base value = 200 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 120 } var:dragon_size_base < 150 }
						set_variable = { name = dragon_size_base value = 150 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 100 } var:dragon_size_base < 120 }
						set_variable = { name = dragon_size_base value = 120 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 80 } var:dragon_size_base < 90 }
						set_variable = { name = dragon_size_base value = 90 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 65 } var:dragon_size_base < 60 }
						set_variable = { name = dragon_size_base value = 60 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 55 } var:dragon_size_base < 45 }
						set_variable = { name = dragon_size_base value = 45 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 45 } var:dragon_size_base < 30 }
						set_variable = { name = dragon_size_base value = 30 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 35 } var:dragon_size_base < 20 }
						set_variable = { name = dragon_size_base value = 20 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 25 } var:dragon_size_base < 15 }
						set_variable = { name = dragon_size_base value = 15 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 20 } var:dragon_size_base < 10 }
						set_variable = { name = dragon_size_base value = 10 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 15 } var:dragon_size_base < 6 }
						set_variable = { name = dragon_size_base value = 6 }
					}
					else_if = {
						limit = { scope:dob_bday_d = { age >= 5 } var:dragon_size_base < 3 }
						set_variable = { name = dragon_size_base value = 3 }
					}
				}
			}
			# Atualizar modifier do rider se tiver um
			if = {
				limit = { has_variable = current_rider exists = var:current_rider }
				var:current_rider = {
					if = {
						# FIX: is_alive = yes guards against ghost/weak references
						limit = { exists = var:current_dragon  var:current_dragon = { is_alive = yes } }
						var:current_dragon = { dob_update_dragon_power_modifier = yes }
					}
					dob_update_rider_dragon_modifier = yes
				}
			}
		}
		# ── Skybreaker: todo dragão que atingir Tier 10 ganha a habilidade ──────────
		# Tier 10 = dragon_size >= 300. permanent = yes: não expira.
		if = {
			limit = {
				has_trait = dragon
				dragon_size >= 300
				NOT = { has_character_modifier = dob_dragon_skybreaker }
			}
			add_character_modifier = dob_dragon_skybreaker
		}
	}
}
