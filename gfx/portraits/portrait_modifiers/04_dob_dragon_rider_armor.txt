clothes_armor = {

	usage = game
	selection_behavior = weighted_random
	priority = 4

	# ── Dragon Rider's Plate: Body + Legwear (steel+gold recolor) ───────────────
	dob_clothing_dragon_rider_armor = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = clothes
				template = agot_all_armors
				accessory = dob_dragon_rider_armor_body
			}
			accessory = {
				mode = add
				gene = legwear
				template = agot_all_legwear
				accessory = dob_dragon_rider_armor_legwear
			}
			morph = {
				mode = modify_multiply
				gene = gene_bs_bust
				value = 0.8 # Reduce exaggerated bust effect when wearing armor
				template = bust_clothes
			}
		}
		outfit_tags = { military_outfit }
		weight = {
			base = 0
			modifier = {
				add = agot_artifact_portrait_modifier  # 500 — ensures override
				exists = this
				portrait_wear_armor_trigger = yes
				agot_has_artifact_equipped = { ARTIFACT_VARIABLE = dob_dragon_rider_armor_artifact }
			}
		}
	}
}
