###############################################################
# DRAGON ON BATTLE - Portrait Modifiers: Dragon Rider's Plate
#
# Body + Legwear → clothes_armor (priority 4)
# Headgear       → headgear_armor (priority 4), SEM portrait_wear_helmet_trigger
#
# SEM portrait_wear_helmet_trigger: a trigger retorna no na janela do
# personagem, então outros mods ganham a seleção. Sem ela, nosso entry
# tem peso 1000000 sempre que o artefato estiver equipado — outros entries
# de headgear_armor têm peso 0 fora de batalha (dependem da trigger),
# então perdemos para ninguém.
###############################################################

# ── Body + Legwear ────────────────────────────────────────────────────────────
clothes_armor = {

	usage = game
	selection_behavior = weighted_random
	priority = 4

	dob_clothing_dragon_rider_armor = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = clothes
				template = agot_all_armors
				accessory = dob_dragon_rider_armor_body
			}
			accessory = {
				mode = add
				gene = legwear
				template = agot_all_legwear
				accessory = dob_dragon_rider_armor_legwear
			}
			morph = {
				mode = modify_multiply
				gene = gene_bs_bust
				value = 0.8
				template = bust_clothes
			}
		}
		outfit_tags = { military_outfit }
		weight = {
			base = 0
			modifier = {
				add = 1000000
				exists = this
				agot_has_artifact_equipped = { ARTIFACT_VARIABLE = dob_dragon_rider_armor_artifact }
			}
		}
	}
}

# ── Headgear override ─────────────────────────────────────────────────────────
# Grupo com nome PRÓPRIO em priority = 5, acima do headgear_armor de todos
# os outros mods (AGOT, Armor of the Kingsguard, Crowns, etc. usam priority 4).
# CK3 aplica grupos em ordem crescente de priority: o nosso roda por ÚLTIMO
# e sobrescreve qualquer headgear já selecionado pelos grupos de priority 4.
# Quando o artefato NÃO está equipado, nosso entry tem peso 0 e não seleciona
# nada — o headgear do priority 4 permanece (comportamento correto).
dob_headgear_override = {

	usage = game
	selection_behavior = weighted_random
	priority = 5

	dob_headgear_dragon_rider_armor = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = headgear
				template = agot_most_headgears
				accessory = dob_dragon_rider_armor_headgear
			}
		}
		outfit_tags = { military_outfit }
		weight = {
			base = 0
			modifier = {
				add = 1000000
				exists = this
				agot_has_artifact_equipped = { ARTIFACT_VARIABLE = dob_dragon_rider_armor_artifact }
			}
		}
	}
}
